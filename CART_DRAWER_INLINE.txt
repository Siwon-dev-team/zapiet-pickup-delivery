{%- enddoc -%}

<script
  src="{{ 'cart-drawer.js' | asset_url }}"
  type="module"
  fetchpriority="low"
></script>

<cart-drawer-component
  class="cart-drawer"
  {{ block.shopify_attributes }}
  {% if settings.auto_open_cart_drawer %}
    auto-open
  {% endif %}
>
  <button
    class="button header-actions__action button-unstyled"
    on:click="/open"
    aria-haspopup="dialog"
    aria-label="{{ 'accessibility.cart' | t }}"
    aria-describedby="cart-bubble-text"
    data-testid="cart-drawer-trigger"
  >
    {% render 'cart-icon-component' %}
  </button>

  <dialog
    ref="dialog"
    class="cart-drawer__dialog dialog-modal dialog-drawer color-{{ settings.drawer_color_scheme }}{% if cart.empty? %} cart-drawer--empty{%endif%}"
    aria-labelledby="{% if cart.empty? %}cart-drawer-heading-empty{% else %}cart-drawer-heading{% endif %}"
    scroll-lock
  >
    <div class="cart-drawer__inner">
      <cart-items-component
        class="cart-items-component"
        data-section-id="{{ section.id }}"
      >
        {%- if cart.empty? -%}
          <div class="cart-drawer__header">
            <button
              ref="closeButton"
              on:click="cart-drawer-component/close"
              class="button close-button cart-drawer__close-button button-unstyled"
              aria-label="{{ 'actions.close_dialog' | t }}"
            >
              <span class="svg-wrapper">
                {{- 'icon-close.svg' | inline_asset_content -}}
              </span>
            </button>
          </div>

          <div
            class="cart-drawer__content motion-reduce"
            aria-label="{{ 'accessibility.cart' | t }}"
          >
            <h2
              class="cart-drawer__heading h3 cart-drawer__heading--empty"
              id="cart-drawer-heading-empty"
            >
              {{ 'content.your_cart_is_empty' | t }}
            </h2>

            <div class="cart-drawer__items">
              {% render 'cart-products', drawer_context: 'drawer' %}
            </div>
          </div>
        {%- else -%}
          <div
            class="cart-drawer__header"
            id="cart-drawer-header"
          >
            <h2
              class="cart-drawer__heading h3"
              id="cart-drawer-heading"
            >
              {{ 'content.cart_title' | t }}
              {% render 'cart-bubble' %}
            </h2>

            <button
              ref="closeButton"
              on:click="cart-drawer-component/close"
              class="button close-button cart-drawer__close-button button-unstyled"
              aria-label="{{ 'actions.close_dialog' | t }}"
            >
              <span class="svg-wrapper">
                {{- 'icon-close.svg' | inline_asset_content -}}
              </span>
            </button>
          </div>

          <div
            class="cart-drawer__content motion-reduce"
            aria-label="{{ 'accessibility.cart' | t }}"
            style="--header-height: 60px;"
          >
            <scroll-hint
              class="cart-drawer__items"
            >
              {% render 'cart-products', drawer_context: 'drawer' %}
            </scroll-hint>

            <div
              class="cart-drawer__summary"
            >
              {%- comment -%} Zapiet Pickup & Delivery Widget START {%- endcomment -%}
              <div id="zapiet-cart-widget" style="padding: 15px; border-bottom: 1px solid #e5e7eb; margin-bottom: 15px;">
                <h4 style="margin: 0 0 12px 0; font-size: 16px; font-weight: 600;">Pickup & Delivery</h4>
                <div id="zapiet-widget-content" style="font-size: 14px; color: #6b7280;">Loading options...</div>
              </div>
              
              <script>
              (function() {
                const shop = '{{ shop.permanent_domain }}';
                const cartTotal = {{ cart.total_price | divided_by: 100.0 }};
                const cartWeight = {{ cart.total_weight | divided_by: 1000.0 }};
                
                fetch(`/apps/zapiet?shop=${shop}`)
                  .then(r => r.json())
                  .then(data => {
                    const content = document.getElementById('zapiet-widget-content');
                    if (!content) return;
                    
                    const { settings, locations } = data;
                    let html = '<div style="display: flex; gap: 8px; margin-bottom: 12px;">';
                    
                    if (settings.enablePickup) {
                      html += '<button onclick="zapietShowPickup()" style="flex: 1; padding: 10px; background: #008060; color: white; border: none; border-radius: 6px; cursor: pointer; font-weight: 600;">Store Pickup</button>';
                    }
                    if (settings.enableDelivery) {
                      html += '<button onclick="zapietShowDelivery()" style="flex: 1; padding: 10px; background: #008060; color: white; border: none; border-radius: 6px; cursor: pointer; font-weight: 600;">Local Delivery</button>';
                    }
                    html += '</div>';
                    html += '<div id="zapiet-options"></div>';
                    
                    content.innerHTML = html;
                    
                    window.zapietData = data;
                    window.zapietShowPickup = function() {
                      const optionsDiv = document.getElementById('zapiet-options');
                      let opts = '<p style="font-weight: 500; margin: 12px 0 8px 0;">Select a location:</p>';
                      
                      locations.filter(l => l.isPickup).forEach(loc => {
                        opts += `<div style="padding: 12px; border: 1px solid #e5e7eb; border-radius: 6px; margin-bottom: 8px; cursor: pointer;" onclick="zapietSelectLocation('${loc.name}', 'Pickup')">
                          <div style="font-weight: 600;">${loc.name}</div>
                          <div style="font-size: 13px; color: #6b7280;">${loc.address}${loc.city ? ', ' + loc.city : ''}</div>
                        </div>`;
                      });
                      
                      optionsDiv.innerHTML = opts;
                    };
                    
                    window.zapietShowDelivery = function() {
                      const optionsDiv = document.getElementById('zapiet-options');
                      optionsDiv.innerHTML = `
                        <p style="font-weight: 500; margin: 12px 0 8px 0;">Enter postal code:</p>
                        <div style="display: flex; gap: 8px;">
                          <input type="text" id="zapiet-postal" placeholder="e.g., V5R" style="flex: 1; padding: 10px; border: 1px solid #d1d5db; border-radius: 6px;" />
                          <button onclick="zapietCheckPostal()" style="padding: 10px 20px; background: #008060; color: white; border: none; border-radius: 6px; cursor: pointer;">Check</button>
                        </div>
                        <div id="zapiet-delivery-result" style="margin-top: 12px;"></div>
                      `;
                    };
                    
                    window.zapietCheckPostal = function() {
                      const postal = document.getElementById('zapiet-postal').value;
                      const resultDiv = document.getElementById('zapiet-delivery-result');
                      if (postal.length >= 3) {
                        resultDiv.innerHTML = '<div style="padding: 10px; background: #ecfdf5; border: 1px solid #10b981; border-radius: 6px; color: #065f46;">✓ Delivery available to ' + postal.toUpperCase() + '!</div>';
                        zapietSelectLocation('Delivery to ' + postal, 'Delivery');
                      } else {
                        resultDiv.innerHTML = '<div style="padding: 10px; background: #fef2f2; border: 1px solid #fecaca; border-radius: 6px; color: #991b1b;">Please enter a valid postal code</div>';
                      }
                    };
                    
                    window.zapietSelectLocation = function(location, method) {
                      fetch('/cart/update.js', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                          attributes: {
                            'Method': method,
                            'Pickup Location': location
                          }
                        })
                      }).then(() => {
                        document.getElementById('zapiet-options').innerHTML = '<div style="padding: 12px; background: #ecfdf5; border-radius: 6px; color: #065f46; font-weight: 500;">✓ ' + method + ' selected: ' + location + '</div>';
                      });
                    };
                  })
                  .catch(err => {
                    console.error('Zapiet widget error:', err);
                    document.getElementById('zapiet-widget-content').innerHTML = '<div style="color: #991b1b;">Unable to load options</div>';
                  });
              })();
              </script>
              {%- comment -%} Zapiet Widget END {%- endcomment -%}
              
              {% render 'cart-summary' %}
            </div>
          </div>
        {%- endif -%}
      </cart-items-component>
    </div>
  </dialog>
</cart-drawer-component>
